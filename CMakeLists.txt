cmake_minimum_required(VERSION 3.21)
project(StrixLookahead VERSION 0.4.1 LANGUAGES C CXX)

include(GNUInstallDirs)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    set(_sl_build_tests_default ${BUILD_TESTING})
else()
    # Consumers opt into our tests without changing their BUILD_TESTING policy.
    set(_sl_build_tests_default OFF)
endif()
option(SL_ENABLE_HIP "Build the Strix Halo HIP backend" OFF)
option(SL_ENABLE_CUDA "Build the CUDA backend" OFF)
option(SL_BUILD_EXAMPLES "Build examples and component benchmarks" ${PROJECT_IS_TOP_LEVEL})
option(SL_BUILD_TESTS "Build Lookahead component contract tests" ${_sl_build_tests_default})
option(SL_ENABLE_INSTALL "Install the standalone Lookahead C ABI package" ${PROJECT_IS_TOP_LEVEL})
unset(_sl_build_tests_default)

add_library(strix_lookahead STATIC src/cache.cpp src/pair.cpp src/selection.cpp)
add_library(StrixLookahead::cache ALIAS strix_lookahead)
set_target_properties(strix_lookahead PROPERTIES
    EXPORT_NAME cache CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_compile_features(strix_lookahead PUBLIC cxx_std_20)
target_include_directories(strix_lookahead PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(strix_lookahead PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
elseif(MSVC)
    target_compile_options(strix_lookahead PRIVATE /W4 /fp:strict)
endif()

set(SL_INSTALL_TARGETS strix_lookahead)
if(SL_ENABLE_HIP)
    if(NOT DEFINED CMAKE_HIP_ARCHITECTURES)
        set(CMAKE_HIP_ARCHITECTURES gfx1151 CACHE STRING "HIP GPU architectures")
    endif()
    enable_language(HIP)
    add_library(strix_lookahead_hip STATIC src/device_hip.hip)
    add_library(StrixLookahead::hip ALIAS strix_lookahead_hip)
    set_target_properties(strix_lookahead_hip PROPERTIES EXPORT_NAME hip
        HIP_STANDARD 20 HIP_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
    target_compile_options(strix_lookahead_hip PRIVATE -ffp-contract=off -fno-fast-math -fno-gpu-flush-denormals-to-zero)
    target_link_libraries(strix_lookahead_hip PUBLIC StrixLookahead::cache)
    if(DEFINED SL_HIP_INCLUDE_DIR OR DEFINED SL_HIP_RUNTIME_LIBRARY)
        # Source embedding can reuse a parent's SDK without hipConfig.cmake.
        # Its absolute SDK paths are deliberately not an installable package.
        if(NOT EXISTS "${SL_HIP_INCLUDE_DIR}/hip/hip_runtime.h" OR NOT EXISTS "${SL_HIP_RUNTIME_LIBRARY}")
            message(FATAL_ERROR "Supply both a valid SL_HIP_INCLUDE_DIR and SL_HIP_RUNTIME_LIBRARY")
        endif()
        if(SL_ENABLE_INSTALL)
            message(FATAL_ERROR "An embedded HIP runtime requires SL_ENABLE_INSTALL=OFF; use hipConfig.cmake for a standalone install")
        endif()
        target_include_directories(strix_lookahead_hip SYSTEM PUBLIC "${SL_HIP_INCLUDE_DIR}")
        target_compile_definitions(strix_lookahead_hip PUBLIC $<$<COMPILE_LANGUAGE:CXX>:__HIP_PLATFORM_AMD__=1>)
        target_link_libraries(strix_lookahead_hip PUBLIC "${SL_HIP_RUNTIME_LIBRARY}")
    else()
        find_package(hip CONFIG REQUIRED)
        target_link_libraries(strix_lookahead_hip PUBLIC hip::host)
    endif()
    list(APPEND SL_INSTALL_TARGETS strix_lookahead_hip)
endif()

if(SL_ENABLE_CUDA)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES "86;89;120" CACHE STRING "CUDA GPU architectures")
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit 12.8 REQUIRED)
    add_library(strix_lookahead_cuda STATIC src/device_cuda.cu)
    add_library(StrixLookahead::cuda ALIAS strix_lookahead_cuda)
    set_target_properties(strix_lookahead_cuda PROPERTIES EXPORT_NAME cuda
        CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
    target_compile_options(strix_lookahead_cuda PRIVATE --fmad=false --ftz=false)
    target_link_libraries(strix_lookahead_cuda PUBLIC StrixLookahead::cache CUDA::cudart)
    list(APPEND SL_INSTALL_TARGETS strix_lookahead_cuda)
endif()

if(SL_BUILD_TESTS)
    enable_testing()
    add_executable(sl_cache_test tests/cache_test.cpp)
    target_link_libraries(sl_cache_test PRIVATE StrixLookahead::cache)
    add_test(NAME lookahead.cache_contract COMMAND sl_cache_test)
    add_executable(sl_pair_test tests/pair_test.cpp)
    target_link_libraries(sl_pair_test PRIVATE StrixLookahead::cache)
    add_test(NAME lookahead.pair_contract COMMAND sl_pair_test)
    add_executable(sl_selection_test tests/selection_test.cpp)
    target_link_libraries(sl_selection_test PRIVATE StrixLookahead::cache)
    add_test(NAME lookahead.selection_contract COMMAND sl_selection_test)
    add_executable(sl_c_api_smoke tests/c_api_smoke.c)
    set_target_properties(sl_c_api_smoke PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
    target_link_libraries(sl_c_api_smoke PRIVATE StrixLookahead::cache)
    add_test(NAME lookahead.c_abi_smoke COMMAND sl_c_api_smoke)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME lookahead.evidence_tools_contract
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/evidence_tools_test.py)
        add_test(NAME lookahead.engine_qualification_contract
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_qualification_test.py)
        add_test(NAME lookahead.predecessor_trace_contract
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/predecessor_trace_test.py)
    endif()
endif()

if(SL_BUILD_EXAMPLES)
    add_executable(sl_cache_example examples/cache_example.cpp)
    target_link_libraries(sl_cache_example PRIVATE StrixLookahead::cache)
    add_executable(sl_pair_example examples/pair_example.cpp)
    target_link_libraries(sl_pair_example PRIVATE StrixLookahead::cache)
    add_executable(sl_pair_benchmark examples/pair_benchmark.cpp)
    target_link_libraries(sl_pair_benchmark PRIVATE StrixLookahead::cache)
endif()

foreach(SL_BACKEND IN ITEMS hip cuda)
    string(TOUPPER ${SL_BACKEND} SL_BACKEND_UPPER)
    if(SL_ENABLE_${SL_BACKEND_UPPER})
        if(SL_BUILD_TESTS)
            add_executable(sl_${SL_BACKEND}_test tests/device_test.cpp)
            target_compile_definitions(sl_${SL_BACKEND}_test PRIVATE SL_TEST_${SL_BACKEND_UPPER}=1)
            target_link_libraries(sl_${SL_BACKEND}_test PRIVATE StrixLookahead::${SL_BACKEND})
            add_test(NAME lookahead.${SL_BACKEND}_device_contract COMMAND sl_${SL_BACKEND}_test)
            set_tests_properties(lookahead.${SL_BACKEND}_device_contract PROPERTIES SKIP_RETURN_CODE 77)
        endif()
        if(SL_BUILD_EXAMPLES)
            add_executable(sl_${SL_BACKEND}_benchmark examples/device_benchmark.cpp)
            target_compile_definitions(sl_${SL_BACKEND}_benchmark PRIVATE SL_TEST_${SL_BACKEND_UPPER}=1)
            target_link_libraries(sl_${SL_BACKEND}_benchmark PRIVATE StrixLookahead::${SL_BACKEND})
        endif()
    endif()
endforeach()

if(SL_ENABLE_INSTALL)
    include(CMakePackageConfigHelpers)
    install(TARGETS ${SL_INSTALL_TARGETS} EXPORT StrixLookaheadTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
    install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/StrixLookahead)
    install(EXPORT StrixLookaheadTargets NAMESPACE StrixLookahead:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StrixLookahead)
    configure_package_config_file(cmake/StrixLookaheadConfig.cmake.in
        ${CMAKE_CURRENT_BINARY_DIR}/StrixLookaheadConfig.cmake
        INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StrixLookahead)
    write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/StrixLookaheadConfigVersion.cmake
        VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/StrixLookaheadConfig.cmake
        ${CMAKE_CURRENT_BINARY_DIR}/StrixLookaheadConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StrixLookahead)
endif()
